309 def check_telegram_authorization(auth_data: dict) -> bool:
310     bot_token = os.environ.get("BOT_TOKEN")
311     if not bot_token:
312         return False
313     check_hash = auth_data.pop('hash', None)
314     if not check_hash:
315         return False
316     
317     if time.time() - int(auth_data.get('auth_date', 0)) > 86400:
318         return False
319         
320     data_check_arr = [f"{key}={value}" for key, value in auth_data.items() if value]
321     data_check_arr.sort()
322     data_check_string = "\n".join(data_check_arr)
323     
324     secret_key = hashlib.sha256(bot_token.encode()).digest()
325     hash_computed = hmac.new(secret_key, data_check_string.encode(), hashlib.sha256).hexdigest()
326     
327     return hmac.compare_digest(hash_computed, check_hash)

358 def check_webapp_authorization(init_data: str) -> bool:
359     import urllib.parse
360     bot_token = os.environ.get("BOT_TOKEN") or os.environ.get("TG_BOT_TOKEN")
361     if not bot_token:
362         return False
363         
364     parsed_data = urllib.parse.parse_qsl(init_data)
365     auth_data = dict(parsed_data)
366     
367     check_hash = auth_data.pop('hash', None)
368     if not check_hash:
369         return False
370         
371     data_check_arr = [f"{key}={value}" for key, value in auth_data.items()]
372     data_check_arr.sort()
373     data_check_string = "\n".join(data_check_arr)
374     
375     secret_key = hmac.new(b"WebAppData", bot_token.encode(), hashlib.sha256).digest()
376     hash_computed = hmac.new(secret_key, data_check_string.encode(), hashlib.sha256).hexdigest()
377     
378     return hmac.compare_digest(hash_computed, check_hash)

435             session.permanent = True

533 def remnawave_create_or_extend_user(username, expire_date_str):
534     api_key = os.environ.get("RW_API_KEY")
535     if not api_key:
536         return None
537         
538     headers = {
539         "Authorization": f"Bearer {api_key}",
540         "Content-Type": "application/json"
541     }
542     
543     # expire_date_str is YYYY-MM-DD HH:MM:SS
544     expire_dt = datetime.strptime(expire_date_str, "%Y-%m-%d %H:%M:%S")
545     # Remnawave expects ISO 8601 with Z
546     expire_iso = expire_dt.strftime("%Y-%m-%dT%H:%M:%S.000Z")
547     
548     # Check if user already exists
549     users_resp = requests.get("https://panel.jointhevoid.ru/api/users", headers=headers, timeout=5)
550     existing_user = None
551     if users_resp.status_code == 200:
552         rw_data = users_resp.json().get("response", {})
553         rw_users = rw_data.get("users", []) if isinstance(rw_data, dict) else []
554         for u in rw_users:
555             if isinstance(u, dict) and u.get("username") == username:
556                 existing_user = u
557                 break
558                 
559     if existing_user:
560         # Extend user
561         update_payload = {
562             "status": "ACTIVE",
563             "expireAt": expire_iso
564         }
565         user_uuid = existing_user["uuid"]
566         resp = requests.put(f"https://panel.jointhevoid.ru/api/users/{user_uuid}", headers=headers, json=update_payload, timeout=5)
567         if resp.status_code == 200:
568             return existing_user.get("subscriptionUrl")
569     else:
570         # Create user
571         squad_uuid = get_remnawave_squad_uuid(api_key)
572         create_payload = {
573             "username": username,
574             "status": "ACTIVE",
575             "trafficLimitBytes": 0,
576             "trafficLimitStrategy": "NO_RESET",
577             "expireAt": expire_iso,
578             "activeInternalSquads": [squad_uuid]
579         }
580         resp = requests.post("https://panel.jointhevoid.ru/api/users", headers=headers, json=create_payload, timeout=5)
581         if resp.status_code == 201:
582             data = resp.json().get("response", {})
583             return data.get("subscriptionUrl")
584             
585     return None

620 def add_subscription(user_id, days_to_add):
621     user = query_one("SELECT * FROM users WHERE id = ?", (user_id,))
622     if not user:
623         return
624         
625     now = datetime.now()
626     if user["expires_at"]:
627         try:
628             current_expires = datetime.strptime(user["expires_at"], "%Y-%m-%d %H:%M:%S")
629             if current_expires > now:
630                 new_expires = current_expires + timedelta(days=days_to_add)
631             else:
632                 new_expires = now + timedelta(days=days_to_add)
633         except ValueError:
634             new_expires = now + timedelta(days=days_to_add)
635     else:
636         new_expires = now + timedelta(days=days_to_add)
637         
638     expires_str = new_expires.strftime("%Y-%m-%d %H:%M:%S")
639     
640     cyrillic_translit = {

910         params.extend([f"%{q}%", f"%{q}%", q])

913     
914     users = [dict(u) for u in query_all(query, params)]
